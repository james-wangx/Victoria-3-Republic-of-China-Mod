	flavored_company = yes

	preferred_headquarters = { STATE_EASTERN_HUBEI }

	building_types = {
		building_coal_mine
		building_iron_mine
		building_steel_mill
	}

	extension_building_types = {
		building_railway
	}

	possible_prestige_goods = {
		prestige_good_generic_steel
	}

	potential = {
		OR = {
			country_has_primary_culture = cu:han
			country_has_primary_culture = cu:manchu
		}
	}

	attainable = {
		hidden_trigger = { # Hide attainable if it's always fulfilled by possible
			any_scope_state = {
				is_incorporated = yes
			}
		}
	}

	possible = {
		has_technology_researched = open_hearth_process
		any_scope_state = {
			is_incorporated = yes
			any_scope_building = {
				is_building_type = building_steel_mill
				level >= 5
			}
		}
	}

	prosperity_modifier = {
		building_steel_mill_throughput_add = 0.10
		building_railway_throughput_add = 0.10
	}

	ai_will_do = {
		has_technology_researched = open_hearth_process
		any_scope_state = {
			is_incorporated = yes
		}
	}

	ai_construction_targets = {
		building_steel_mill = {
			level = 5
			state_trigger = {
				is_incorporated = yes
			}
		}
	}

	ai_weight = {
		value = 3 # Higher base value for flavored companies
	}
}
